cmake_minimum_required(VERSION 3.24)
project(KeyboardTest VERSION 1.0.10.16 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "KeyboardTest requires Windows, MSVC and x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(FTC_FLUENTUI_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../FluentUI/sdk" CACHE PATH "Native FluentUI SDK")
find_package(FluentUI 0.2 CONFIG REQUIRED HINTS "${FTC_FLUENTUI_SDK_DIR}")
include(CTest)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/EHsc;/permissive-;/utf-8;/guard:cf>")
add_library(ModuleCommunication STATIC src/modules/ModuleCommunication.cpp)
add_library(ModuleConfig STATIC src/modules/ModuleConfig.cpp)
target_include_directories(ModuleCommunication PUBLIC src)
target_include_directories(ModuleConfig PUBLIC src)
target_link_libraries(ModuleConfig PUBLIC bcrypt)
add_subdirectory(src/modules/keyboard_test)
add_subdirectory(src/keyboard_editor)
configure_file(resources/KeyboardTest.rc.in generated/KeyboardTest.rc @ONLY)
configure_file(resources/KeyboardTest.manifest.in generated/KeyboardTest.manifest @ONLY)
add_library(KeyboardApplication STATIC src/app/Application.cpp src/app/Application.h)
target_include_directories(KeyboardApplication PUBLIC src)
target_compile_definitions(KeyboardApplication PRIVATE KEYBOARD_VERSION=L\"${PROJECT_VERSION}\")
target_link_libraries(KeyboardApplication PUBLIC KeyboardTestModule ModuleConfig user32)
add_executable(KeyboardTest WIN32 src/app/Main.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/KeyboardTest.rc")
target_include_directories(KeyboardTest PRIVATE resources "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(KeyboardTest PRIVATE KeyboardApplication shell32)
target_link_options(KeyboardTest PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
if(BUILD_TESTING)
    add_executable(KeyboardApplicationTests tests/ApplicationTests.cpp)
    target_link_libraries(KeyboardApplicationTests PRIVATE KeyboardApplication)
    add_test(NAME KeyboardApplicationTests COMMAND KeyboardApplicationTests)
    set_tests_properties(KeyboardApplicationTests PROPERTIES TIMEOUT 20)
endif()
