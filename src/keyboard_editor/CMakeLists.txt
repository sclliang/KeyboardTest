configure_file(KBeditor.rc.in "${CMAKE_CURRENT_BINARY_DIR}/KBeditor.rc" @ONLY)
configure_file(KBeditor.manifest.in "${CMAKE_CURRENT_BINARY_DIR}/KBeditor.manifest" @ONLY)
add_executable(KBeditor WIN32 KBeditor.cpp KeyboardLayout.cpp KeyboardLogic.cpp KeyboardUi.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/KBeditor.rc")
target_include_directories(KBeditor PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(KBeditor PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_compile_options(KBeditor PRIVATE /W4 /WX /EHsc /permissive- /utf-8 /guard:cf)
target_link_options(KBeditor PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /GUARD:CF /CETCOMPAT)
target_link_libraries(KBeditor PRIVATE fluent::FluentUI d2d1 dwrite dwmapi user32 gdi32 comctl32 comdlg32 shell32)

endif()
